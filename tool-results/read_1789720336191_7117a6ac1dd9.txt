    32→    32→export type DemandLevel = "High" | "Medium" | "Low";
    33→    33→export type SupplyLevel = "Strong" | "Moderate" | "Limited" | "Low Evidence" | "None";
    34→    34→
    35→    35→export interface DemandPulseSkill {
    36→    36→  skillId: string;
    37→    37→  skillName: string;
    38→    38→  demandLevel: DemandLevel;
    39→    39→  opportunityCount: number;
    40→    40→  requiredRoleLevels: { roleId: string; roleName: string; requiredLevel: number }[];
    41→    41→  openRoles: number;
    42→    42→  openChallenges: number;
    43→    43→  talentCoverage: number; // 0–100, computed from candidates
    44→    44→  insufficientCoverage: boolean;
    45→    45→}

    47→    47→export const IndustryAggregator = {
    48→    48→  // §3 Demand Pulse — derived from published opportunities + candidate supply
    49→    49→  getDemandPulse: (): DemandPulseSkill[] => {
    50→    50→    const opps = useCareerStore.getState().opportunities.filter((o) => o.status === "Published");
    51→    51→    const candidates = DEMO_CANDIDATES;
    52→    52→    const challenges = useIndustryStore.getState().challenges;
    53→    53→
    54→    54→    const map = new Map<string, DemandPulseSkill>();
    55→    55→    for (const o of opps) {
    56→    56→      for (const rs of o.requiredSkills) {
    57→    57→        const e = map.get(rs.skillId) ?? {
    58→    58→          skillId: rs.skillId,
    59→    59→          skillName: SKILL_NAMES[rs.skillId] ?? rs.skillId,
    60→    60→          demandLevel: "Low" as DemandLevel,
    61→    61→          opportunityCount: 0,
    62→    62→          requiredRoleLevels: [] as { roleId: string; roleName: string; requiredLevel: number }[],
    63→    63→          openRoles: new Set<string>(),
    64→    64→          openChallenges: 0,
    65→    65→          talentCoverage: 0,
    66→    66→          insufficientCoverage: false,
    67→    67→        };
    68→    68→        e.opportunityCount++;
    69→    69→        o.targetRoles.forEach((r) => (e.openRoles as Set<string>).add(r));
    70→    70→        e.requiredRoleLevels.push({
    71→    71→          roleId: o.targetRoles[0] ?? "—",
    72→    72→          roleName: ALL_ROLES.find((role) => role.roleId === o.targetRoles[0])?.roleName ?? "—",
    73→    73→          requiredLevel: rs.requiredLevel,
    74→    74→        });
    75→    75→        map.set(rs.skillId, e);
    76→    76→      }
    77→    77→    }
    78→    78→    for (const ch of challenges) {
    79→    79→      for (const s of ch.requiredSkills) {
    80→    80→        const e = map.get(s.skillId);
    81→    81→        if (e) e.openChallenges++;
    82→    82→      }
    83→    83→    }
    84→    84→
    85→    85→    const result: DemandPulseSkill[] = [];
    86→    86→    for (const [, e] of map) {
    87→    87→      e.openRoles = (e.openRoles as Set<string>).size;
    88→    88→      e.demandLevel = e.opportunityCount >= 3 ? "High" : e.opportunityCount >= 2 ? "Medium" : "Low";
    89→    89→      // supply: average competency among candidates who have this skill
    90→    90→      const scores = candidates
    91→    91→        .map((c) => c.competencies[e.skillId]?.competencyScore)
    92→    92→        .filter((s): s is number => typeof s === "number");
    93→    93→      e.talentCoverage = scores.length > 0 ? Math.round(scores.reduce((a, b) => a + b, 0) / scores.length) : 0;
    94→    94→      e.insufficientCoverage = e.talentCoverage < 60;
    95→    95→      result.push(e);
    96→    96→    }
    97→    97→    return result.sort((a, b) => b.opportunityCount - a.opportunityCount);
    98→    98→  },

   100→   100→  // §4 What We Need — top demanded role + its critical skills + required levels
   101→   101→  getWhatWeNeed: () => {
   102→   102→    const opps = useCareerStore.getState().opportunities.filter((o) => o.status === "Published");
   103→   103→    const candidates = DEMO_CANDIDATES;
   104→   104→    const roleMap = new Map<string, { roleId: string; roleName: string; criticalSkills: Map<string, { skillId: string; skillName: string; requiredLevel: number; importance: string }[]>; opportunityCount: number }>();
   105→   105→
   106→   106→    for (const o of opps) {
   107→   107→      const roleId = o.targetRoles[0] ?? "—";
   108→   108→      const role = ALL_ROLES.find((r) => r.roleId === roleId);
   109→   109→      const e = roleMap.get(roleId) ?? {
   110→   110→        roleId,
   111→   111→        roleName: role?.roleName ?? roleId,
   112→   112→        criticalSkills: new Map<string, { skillId: string; skillName: string; requiredLevel: number; importance: string }>(),
   113→   113→        opportunityCount: 0,
   114→   114→      };
   115→   115→      e.opportunityCount++;
   116→   116→      for (const rs of o.requiredSkills) {
   117→   117→        e.criticalSkills.set(rs.skillId, {
   118→   118→          skillId: rs.skillId,
   119→   119→          skillName: SKILL_NAMES[rs.skillId] ?? rs.skillId,
   120→   120→          requiredLevel: rs.requiredLevel,
   121→   121→          importance: rs.importance,
   122→   122→        });
   123→   123→      }
   124→   124→      roleMap.set(roleId, e);
   125→   125→    }
   126→   126→
   127→   127→    return [...roleMap.values()]
   128→   128→      .map((r) => {
   129→   129→        const skills = [...r.criticalSkills.values()];
   130→   130→        // Compute talent availability for each critical skill
   131→   131→        const skillAvailability = skills.map((s) => {
   132→   132→          const matchingCandidates = candidates.filter((c) => (c.competencies[s.skillId]?.competencyScore ?? 0) >= s.requiredLevel).length;
   133→   133→          return { ...s, availableCandidates: matchingCandidates, totalCandidatesConsidered: candidates.length };
   134→   134→        });
   135→   135→        const overallAvailability = skillAvailability.reduce((sum, s) => sum + s.availableCandidates, 0);
   136→   136→        return {
   137→   137→          roleId: r.roleId,
   138→   138→          roleName: r.roleName,
   139→   139→          opportunityCount: r.opportunityCount,
   140→   140→          criticalSkills: skillAvailability,
   141→   141→          overallAvailability,
   142→   142→          insufficient: overallAvailability < skills.length,
   143→   143→        };
   144→   144→      })
   145→   145→      .sort((a, b) => b.opportunityCount - a.opportunityCount);
   146→   146→  },

   148→   148→  // §5 Talent Supply — what the talent pool can demonstrate per skill
   149→   149→  getTalentSupply: () => {
   150→   150→    const candidates = DEMO_CANDIDATES;
   151→   151→    const opps = useCareerStore.getState().opportunities.filter((o) => o.status === "Published");
   152→   152→    const demandedSkillIds = new Set<string>();
   153→   153→    for (const o of opps) for (const rs of o.requiredSkills) demandedSkillIds.add(rs.skillId);
   154→   154→
   155→   155→    const supplyMap = new Map<string, { skillId: string; skillName: string; scores: number[]; roles: Set<string> }>();
   156→   156→    for (const c of candidates) {
   157→   157→      for (const [, comp] of Object.entries(c.competencies)) {
   158→   158→        const e = supplyMap.get(comp.skillId) ?? { skillId: comp.skillId, skillName: SKILL_NAMES[comp.skillId] ?? comp.skillId, scores: [], roles: new Set<string>() };
   159→   159→        e.scores.push(comp.competencyScore);
   160→   160→        e.roles.add(c.targetRole);
   161→   161→        supplyMap.set(comp.skillId, e);
   162→   162→      }
   163→   163→    }
   164→   164→
   165→   165→    const supply = [...supplyMap.entries()].map(([skillId, e]) => {
   166→   166→      const avg = e.scores.length > 0 ? Math.round(e.scores.reduce((a, b) => a + b, 0) / e.scores.length) : 0;
   167→   167→      const supplyLevel: SupplyLevel = avg >= 70 ? "Strong" : avg >= 50 ? "Moderate" : avg >= 30 ? "Limited" : avg > 0 ? "Low Evidence" : "None";
   168→   168→      return {
   169→   169→        skillId,
   170→   170→        skillName: e.skillName,
   171→   171→        avgCompetency: avg,
   172→   172→        supplyLevel,
   173→   173→        demonstratedBy: e.scores.length,
   174→   174→        targetRoles: [...e.roles],
   175→   175→        inDemand: demandedSkillIds.has(skillId),
   176→   176→      };
   177→   177→    });
   178→   178→    return supply.sort((a, b) => (a.inDemand === b.inDemand ? b.avgCompetency - a.avgCompetency : a.inDemand ? -1 : 1));
   179→   179→  },

   181→   181→  // §12 Opportunity Intelligence — projected talent availability before publishing
   182→   182→  getOpportunityIntelligence: (requiredSkills: { skillId: string; requiredLevel: number }[]) => {
   183→   183→    const candidates = DEMO_CANDIDATES;
   184→   184→    const perSkill = requiredSkills.map((rs) => {
   185→   185→      const matching = candidates.filter((c) => (c.competencies[rs.skillId]?.competencyScore ?? 0) >= rs.requiredLevel);
   186→   186→      const coverage = candidates.length > 0 ? Math.round((matching.length / candidates.length) * 100) : 0;
   187→   187→      return {
   188→   188→        skillId: rs.skillId,
   189→   189→        skillName: SKILL_NAMES[rs.skillId] ?? rs.skillId,
   190→   190→        requiredLevel: rs.requiredLevel,
   191→   191→        matchingCandidates: matching.length,
   192→   192→        totalCandidates: candidates.length,
   193→   193→        coverage,
   194→   194→        limited: matching.length < 2,
   195→   195→      };
   196→   196→    });
   197→   197→    const overallCoverage = perSkill.length > 0 ? Math.round(perSkill.reduce((a, b) => a + b.coverage, 0) / perSkill.length) : 0;
   198→   198→    const suggestion =
   199→   199→      overallCoverage >= 50 ? "Proceed with publishing — sufficient talent availability."
   200→   200→      : overallCoverage >= 25 ? "Consider also creating a project or challenge pipeline to widen the funnel."
   201→   201→      : "Talent availability is limited. Consider a project, challenge, or training pipeline before publishing.";
   202→   202→    return { perSkill, overallCoverage, suggestion, limited: overallCoverage < 25 };
   203→   203→  },

   205→   205→  // §14 Team Builder — skill coverage for a multi-role team spec
   206→   206→  getTeamBuilderCoverage: (teamSpec: { roleIds: string[] }) => {
   207→   207→    const candidates = DEMO_CANDIDATES;
   208→   208→    const requiredSkills = new Map<string, { skillId: string; skillName: string; requiredLevel: number; sourceRole: string }>();
   209→   209→    for (const roleId of teamSpec.roleIds) {
   210→   210→      const role = ALL_ROLES.find((r) => r.roleId === roleId);
   211→   211→      if (!role) continue;
   212→   212→      for (const ws of role.weightedSkills) {
   213→   213→        const e = requiredSkills.get(ws.skillId) ?? { skillId: ws.skillId, skillName: SKILL_NAMES[ws.skillId] ?? ws.skillId, requiredLevel: 0, sourceRole: roleId };
   214→   214→        e.requiredLevel = Math.max(e.requiredLevel, ws.requiredLevel);
   215→   215→        requiredSkills.set(ws.skillId, e);
   216→   216→      }
   217→   217→    }
   218→   218→    const requiredList = [...requiredSkills.values()];
   219→   219→    const coverage = requiredList.map((rs) => {
   220→   220→      const candidatesWith = candidates.filter((c) => (c.competencies[rs.skillId]?.competencyScore ?? 0) >= rs.requiredLevel);
   221→   221→      return { ...rs, availableCandidates: candidatesWith.length, covered: candidatesWith.length > 0 };
   222→   222→    });
   223→   223→    const missing = coverage.filter((c) => !c.covered);
   224→   224→    return {
   225→   225→      requiredSkills: coverage,
   226→   226→      coveragePercent: requiredList.length > 0 ? Math.round((coverage.length - missing.length) / requiredList.length * 100) : 0,
   227→   227→      missing,
   228→   228→      candidateFits: candidates.map((c) => ({
   229→   229→        candidate: c,
   230→   230→        matchCount: coverage.filter((cs) => (c.competencies[cs.skillId]?.competencyScore ?? 0) >= cs.requiredLevel).length,
   231→   231→      })).sort((a, b) => b.matchCount - a.matchCount),
   232→   232→    };
   233→   233→  },

   371→   371→  // §31 Mentor Matching — student gap × faculty expertise (DEMO_FACULTY is a single profile)
   372→   372→  getMentorMatches: () => {
   373→   373→    const candidates = DEMO_CANDIDATES;
   374→   374→    const faculty = AcademiaService.getFaculty();
   375→   375→    const matches: { student: Candidate; gapSkills: string[]; faculty: typeof faculty | null; reason: string }[] = [];
   376→   376→    for (const c of candidates) {
   377→   377→      // Find skills where student is below target (gap)
   378→   378→      const role = ALL_ROLES.find((r) => r.roleId === c.roleId);
   379→   379→      if (!role) continue;
   380→   380→      const gapSkills = role.weightedSkills
   381→   381→        .filter((ws) => (c.competencies[ws.skillId]?.competencyScore ?? 0) < ws.requiredLevel)
   382→   382→        .map((ws) => ws.skillId);
   383→   383→      if (gapSkills.length === 0) continue;
   384→   384→      // Match to faculty with expertise in any gap skill (by skill ID via faculty.skills)
   385→   385→      const facultyHasExpertise = faculty.skills.some((fs) => gapSkills.includes(fs.skillId));
   386→   386→      const matchedSkillNames = faculty.skills
   387→   387→        .filter((fs) => gapSkills.includes(fs.skillId))
   388→   388→        .map((fs) => fs.skillName);
   389→   389→      matches.push({
   390→   390→        student: c,
   391→   391→        gapSkills: gapSkills.map((id) => SKILL_NAMES[id] ?? id),
   392→   392→        faculty: facultyHasExpertise ? faculty : null,
   393→   393→        reason: facultyHasExpertise
   394→   394→          ? `Faculty ${faculty.name} has expertise in ${matchedSkillNames.join(", ")}.`
   395→   395→          : `No faculty expertise currently mapped for ${gapSkills.map((id) => SKILL_NAMES[id] ?? id).join(", ")}.`,
   396→   396→      });
   397→   397→    }
   398→   398→    return matches;
   399→   399→  },
   400→   400→};

   419→   419→  // §53 Intelligence Alerts — derived from demand-supply, gaps, practical exposure
   420→   420→  getAlerts: (): IntelligenceAlert[] => {
   421→   421→    const ds = useInstitutionStore.getState();
   422→   422→    const opps = useCareerStore.getState().opportunities.filter((o) => o.status === "Published");
   423→   423→    const candidates = DEMO_CANDIDATES;
   424→   424→    const alignments = AcademiaService.getCurriculumAlignment();
   425→   425→    const alerts: IntelligenceAlert[] = [];
   426→   426→
   427→   427→    // Critical Skill Gap — demand-supply gap >= 30
   428→   428→    for (const a of alignments) {
   429→   429→      if (a.alignment === "Gap") {
   430→   430→        alerts.push({
   431→   431→          id: `alert-gap-${a.skillId}`,
   432→   432→          tone: "critical",
   433→   433→          category: "Critical Skill Gap",
   434→   434→          what: `${a.skillName} — supply ${a.studentAvgCompetency} vs demand ${a.demandCount} opportunities`,
   435→   435→          why: `Industry demand: ${a.industryDemand}. Curriculum coverage: ${a.curriculumCoverage}. Practical evidence: ${a.practicalEvidence}.`,
   436→   436→          suggestedAction: `Consider intervention: ${a.suggestedEnrichment.join(", ")}`,
   437→   437→          affectedItems: a.targetRoles.length,
   438→   438→        });
   439→   439→      }
   440→   440→    }
   441→   441→
   442→   442→    // Emerging Skill — high demand but not in curriculum
   443→   443→    for (const a of alignments) {
   444→   444→      if (a.industryDemand === "High" && (a.curriculumCoverage === "Not Covered" || a.curriculumCoverage === "Introductory")) {
   445→   445→        alerts.push({
   446→   446→          id: `alert-emerging-${a.skillId}`,
   447→   447→          tone: "emerging",
   448→   448→          category: "Emerging Skill",
   449→   449→          what: `${a.skillName} — emerging industry demand not yet in curriculum`,
   450→   450→          why: `Demand: High (${a.demandCount} opportunities). Current coverage: ${a.curriculumCoverage}.`,
   451→   451→          suggestedAction: "Consider curriculum enrichment and faculty development.",
   452→   452→          affectedItems: a.targetRoles.length,
   453→   453→        });
   454→   454→      }
   455→   455→    }
   456→   456→
   457→   457→    // Low Practical Exposure — curriculum strong but evidence low
   458→   458→    for (const a of alignments) {
   459→   459→      if ((a.curriculumCoverage === "Strong" || a.curriculumCoverage === "Moderate") && (a.practicalEvidence === "Limited" || a.practicalEvidence === "None")) {
   460→   460→        alerts.push({
   461→   461→          id: `alert-practical-${a.skillId}`,
   462→   462→          tone: "warning",
   463→   463→          category: "Low Practical Exposure",
   464→   464→          what: `${a.skillName} — theory covered, but practical evidence is ${a.practicalEvidence.toLowerCase()}`,
   465→   465→          why: `Curriculum: ${a.curriculumCoverage}. Student competency: ${a.studentAvgCompetency}. Verified evidence: limited.`,
   466→   466→          suggestedAction: "Introduce industry project, workshop, or hackathon exposure.",
   467→   467→          affectedItems: a.targetRoles.length,
   468→   468→        });
   469→   469→      }
   470→   470→    }
   471→   471→
   472→   472→    // Low Evidence Confidence — many candidates, few verified evidence
   473→   473→    const evidenceBySkill = new Map<string, { total: number; verified: number }>();
   474→   474→    for (const c of candidates) {
   475→   475→      for (const ev of c.evidence) {
   476→   476→        const e = evidenceBySkill.get(ev.skillId) ?? { total: 0, verified: 0 };
   477→   477→        e.total++; if (ev.status === "Verified") e.verified++;
   478→   478→        evidenceBySkill.set(ev.skillId, e);
   479→   479→      }
   480→   480→    }
   481→   481→    for (const [skillId, e] of evidenceBySkill) {
   482→   482→      if (e.total >= 2 && e.verified === 0) {
   483→   483→        alerts.push({
   484→   484→          id: `alert-confidence-${skillId}`,
   485→   485→          tone: "info",
   486→   486→          category: "Low Evidence Confidence",
   487→   487→          what: `${SKILL_NAMES[skillId] ?? skillId} — ${e.total} evidence records, 0 verified`,
   488→   488→          why: "Evidence has been submitted but not yet verified by industry or faculty.",
   489→   489→          suggestedAction: "Engage industry or faculty to verify outstanding evidence.",
   490→   490→          affectedItems: e.total,
   491→   491→        });
   492→   492→      }
   493→   493→    }
   494→   494→
   495→   495→    // Upcoming Intervention — interventions in Proposed/Recommended state
   496→   496→    for (const inv of ds.interventions) {
   497→   497→      if (inv.status === "Proposed" || inv.status === "Recommended") {
   498→   498→        alerts.push({
   499→   499→          id: `alert-intervention-${inv.id}`,
   500→   500→          tone: "info",
   501→   501→          category: "Upcoming Intervention",
   502→   502→          what: `${inv.type} for ${inv.skillName} — ${inv.status}`,
   503→   503→          why: `Targets: ${inv.skillName}. Affected: ${inv.affectedStudents} student(s). Department: ${inv.department}.`,
   504→   504→          suggestedAction: "Review and approve the intervention to begin execution.",
   505→   505→          affectedItems: inv.affectedStudents,
   506→   506→        });
   507→   507→      }
   508→   508→    }
   509→   509→
   510→   510→    return alerts.slice(0, 12);
   511→   511→  },
   512→   512→
   513→   513→  // §54 What Should The Institution Do Next — prioritized recommendations
   514→   514→  getNextActions: () => {
   515→   515→    const interventions = useInstitutionStore.getState().interventions;
   516→   516→    const cohortInterventions = ((): { skillName: string; suggestedAction: string; priority: string; affectedStudents: number; targetRoles: string[] }[] => {
   517→   517→      // Reuse logic from InstitutionService.getCohortInterventions indirectly
   518→   518→      const opps = useCareerStore.getState().opportunities.filter((o) => o.status === "Published");
   519→   519→      const candidates = DEMO_CANDIDATES;
   520→   520→      const demandMap = new Map<string, { count: number; roles: Set<string> }>();
   521→   521→      for (const o of opps) for (const rs of o.requiredSkills) {
   522→   522→        const e = demandMap.get(rs.skillId) ?? { count: 0, roles: new Set<string>() };
   523→   523→        e.count++; o.targetRoles.forEach((r) => e.roles.add(r));
   524→   524→        demandMap.set(rs.skillId, e);
   525→   525→      }
   526→   526→      const supplyMap = new Map<string, number[]>();
   527→   527→      for (const c of candidates) for (const [, comp] of Object.entries(c.competencies)) {
   528→   528→        const arr = supplyMap.get(comp.skillId) ?? [];
   529→   529→        arr.push(comp.competencyScore); supplyMap.set(comp.skillId, arr);
   530→   530→      }
   531→   531→      const out = [];
   532→   532→      for (const [skillId, { count, roles }] of demandMap) {
   533→   533→        const supply = supplyMap.get(skillId);
   534→   534→        const supplyScore = supply ? Math.round(supply.reduce((a, b) => a + b, 0) / supply.length) : 0;
   535→   535→        const demandScore = Math.min(100, count * 20);
   536→   536→        const gap = demandScore - supplyScore;
   537→   537→        if (gap <= 0) continue;
   538→   538→        const priority = gap >= 40 ? "Critical" : gap >= 25 ? "High" : gap >= 10 ? "Medium" : "Low";
   539→   539→        out.push({
   540→   540→          skillName: SKILL_NAMES[skillId] ?? skillId,
   541→   541→          suggestedAction: priority === "Critical" ? "Industry Workshop" : priority === "High" ? "Live Project" : priority === "Medium" ? "Bootcamp" : "Certification",
   542→   542→          priority,
   543→   543→          affectedStudents: supply?.length ?? 0,
   544→   544→          targetRoles: [...roles],
   545→   545→        });
   546→   546→      }
   547→   547→      return out.sort((a, b) => (a.priority === "Critical" ? 0 : a.priority === "High" ? 1 : 2) - (b.priority === "Critical" ? 0 : b.priority === "High" ? 1 : 2));
   548→   548→    })();
   549→   549→
   550→   550→    return cohortInterventions.slice(0, 5).map((ci, idx) => ({
   551→   551→      rank: idx + 1,
   552→   552→      title: `${ci.suggestedAction} for ${ci.skillName}`,
   553→   553→      reason: `${ci.skillName} gap is ${ci.priority.toLowerCase()}. ${ci.affectedStudents} student(s) currently below target. Relevant roles: ${ci.targetRoles.slice(0, 3).join(", ")}.`,
   554→   554→      suggestedAction: ci.suggestedAction,
   555→   555→      skillName: ci.skillName,
   556→   556→      priority: ci.priority,
   557→   557→    }));
   558→   558→  },

   560→   560→  // §48 Outcome Monitoring — before/after for a given intervention (using evidence as proxy)

   569→   569→    const beforeEvidence = candidates.flatMap((c) => c.evidence.filter((e) => e.skillId === targetSkillId && e.createdAt < startDate));
   570→   570→    const afterEvidence = candidates.flatMap((c) => c.evidence.filter((e) => e.skillId === targetSkillId && e.createdAt >= startDate));
   571→   571→    return {
   572→   572→      intervention: inv,
   573→   573→      before: {
   574→   574→        evidenceCount: beforeEvidence.length,
   575→   575→        verifiedCount: beforeEvidence.filter((e) => e.status === "Verified").length,
   576→   576→      },
   577→   577→      after: {
   578→   578→        evidenceCount: afterEvidence.length,
   579→   579→        verifiedCount: afterEvidence.filter((e) => e.status === "Verified").length,
   580→   580→      },
   581→   581→      observedChange: afterEvidence.length - beforeEvidence.length,
   582→   582→      newEvidenceGenerated: afterEvidence.length,
   583→   583→    };
   584→   584→  },

   628→   628→  // §49 Placement Intelligence — common skills among selected candidates
   629→   629→  getPlacementIntelligence: () => {
   630→   630→    const apps = useCareerStore.getState().applications;
   631→   631→    const opps = useCareerStore.getState().opportunities;
   632→   632→    const selectedCandidates = apps.filter((a) => a.status === "SELECTED").map((a) => a.studentId);
   633→   633→    const candidates = DEMO_CANDIDATES.filter((c) => selectedCandidates.includes(c.studentId));
   634→   634→    if (candidates.length === 0) {
   635→   635→      return {
   636→   636→        targetRoles: [] as string[],
   637→   637→        applicants: apps.length,
   638→   638→        shortlisted: apps.filter((a) => a.status === "SHORTLISTED").length,
   639→   639→        interviews: apps.filter((a) => a.status === "INTERVIEW").length,
   640→   640→        selected: selectedCandidates.length,
   641→   641→        commonSkillsAmongSelected: [] as { skill: string; avgScore: number }[],
   642→   642→      };
   643→   643→    }
   644→   644→    const skillScores = new Map<string, number[]>();
   645→   645→    for (const c of candidates) for (const [, comp] of Object.entries(c.competencies)) {
   646→   646→      const arr = skillScores.get(comp.skillId) ?? [];
   647→   647→      arr.push(comp.competencyScore);
   648→   648→      skillScores.set(comp.skillId, arr);
   649→   649→    }
   650→   650→    const commonSkills = [...skillScores.entries()]
   651→   651→      .map(([id, scores]) => ({ skill: SKILL_NAMES[id] ?? id, avgScore: Math.round(scores.reduce((a, b) => a + b, 0) / scores.length) }))
   652→   652→      .sort((a, b) => b.avgScore - a.avgScore)
   653→   653→      .slice(0, 5);
   654→   654→    const targetRoles = [...new Set(candidates.map((c) => c.targetRole))];
   655→   655→    return {
   656→   656→      targetRoles,
   657→   657→      applicants: apps.length,
   658→   658→      shortlisted: apps.filter((a) => a.status === "SHORTLISTED").length,
   659→   659→      interviews: apps.filter((a) => a.status === "INTERVIEW").length,
   660→   660→      selected: selectedCandidates.length,
   661→   661→      commonSkillsAmongSelected: commonSkills,
   662→   662→    };
   663→   663→  },
   664→   664→};
